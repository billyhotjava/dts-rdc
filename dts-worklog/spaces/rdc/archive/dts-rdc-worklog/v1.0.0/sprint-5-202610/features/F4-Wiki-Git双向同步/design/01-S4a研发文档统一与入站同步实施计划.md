package com.yuzhi.dts.wiki.service.wiki.sync;

import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Map;

/** Local bare-repo helpers for sync integration tests (no SSH, no network). */
final class GitFixtures {

    private GitFixtures() {}

    /** Creates {@code <temp>/<name>.git} seeded with files and a dev clone at {@code <temp>/<name>-dev}. */
    static Path[] remoteWithFiles(Path temp, String name, Map<String, String> files) throws Exception {
        Path remote = temp.resolve(name + ".git");
        sh(temp, "git", "init", "--bare", "-b", "main", remote.toString());
        Path dev = temp.resolve(name + "-dev");
        sh(temp, "git", "clone", remote.toString(), dev.toString());
        sh(dev, "git", "checkout", "-b", "main");
        for (var entry : files.entrySet()) {
            write(dev, entry.getKey(), entry.getValue());
        }
        commitAndPush(dev, "seed");
        return new Path[] { remote, dev };
    }

    static void write(Path dev, String path, String content) throws Exception {
        Path file = dev.resolve(path);
        Files.createDirectories(file.getParent());
        Files.writeString(file, content, StandardCharsets.UTF_8);
    }

    static void commitAndPush(Path dev, String message) throws Exception {
        sh(dev, "git", "add", "-A");
        sh(dev, "git", "-c", "user.name=dev", "-c", "user.email=dev@example.com", "commit", "-m", message);
        sh(dev, "git", "push", "origin", "main");
    }

    static String revParse(Path dir, String rev) throws Exception {
        return sh(dir, "git", "rev-parse", rev).trim();
    }

    static String sh(Path dir, String... args) throws Exception {
        ProcessBuilder builder = new ProcessBuilder(args);
        builder.directory(dir.toFile());
        builder.redirectErrorStream(true);
        Process process = builder.start();
        String output = new String(process.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
        if (process.waitFor() != 0) {
            throw new IllegalStateException("git failed: " + String.join(" ", args) + "\n" + output);
        }
        return output;
    }
}

/** D10 A′: git-bound pages are edited in git, not in the wiki, while outbound sync is off. */
public class GitPageReadOnlyException extends RuntimeException {

    private final String gitPath;

    public GitPageReadOnlyException(String gitPath) {
        super("Page is maintained in git (" + gitPath + "); edit it in the repository");
        this.gitPath = gitPath;
    }

    public String getGitPath() {
        return gitPath;

}

/** Decides whether wiki users may change git-bound pages (D10 A′). */
@Component
public class GitWritePolicy {

    private final WikiProperties properties;

    public GitWritePolicy(WikiProperties properties) {
        this.properties = properties;
    }

    public boolean isGitBound(Page page) {
        return page.getKind() == PageKind.GIT || (page.getKind() == PageKind.FOLDER && page.getGitPath() != null);
    }

    public boolean gitReadOnly(Page page) {
        return !properties.isOutboundEnabled() && isGitBound(page);
    }

    public void requireWritable(Page page) {
        if (gitReadOnly(page)) {
            throw new GitPageReadOnlyException(page.getGitPath());
        }
    }

    /** For pages about to be created or copied into a git-bound location. */
    public void requireWritableTarget(PageKind kind, boolean underSyncRoot, String targetPath) {
        boolean gitTarget = kind == PageKind.GIT || (kind == PageKind.FOLDER && underSyncRoot);
        if (!properties.isOutboundEnabled() && gitTarget) {
            throw new GitPageReadOnlyException(targetPath);
        }
    }
}

/** Creates the spaces declared in {@code application.wiki.spaces} (D17) and their deploy keys. */
@Component
public class SpaceBootstrapper {

    private static final Logger LOG = LoggerFactory.getLogger(SpaceBootstrapper.class);

    private final WikiProperties properties;
    private final SyncAdminService adminService;
    private final SpaceRepository spaceRepository;
    private final SyncRootRepository rootRepository;

    public SpaceBootstrapper(WikiProperties properties, SyncAdminService adminService, SpaceRepository spaceRepository, SyncRootRepository rootRepository) {
        this.properties = properties;
        this.adminService = adminService;
        this.spaceRepository = spaceRepository;
        this.rootRepository = rootRepository;
    }

    @EventListener(ApplicationReadyEvent.class)
    public void onReady() {
        ensureSpaces();
    }

    public void ensureSpaces() {
        for (WikiProperties.SpaceSpec spec : properties.getSpaces()) {
            var existing = spaceRepository.findOneBySlug(spec.getSlug());
            if (existing.isEmpty()) {
                List<SyncAdminService.RootSpec> roots = spec.getRoots().stream().map(SyncAdminService.RootSpec::new).toList();
                adminService.createSpace(spec.getSlug(), spec.getName(), spec.getDescription(), spec.getGitRepoUrl(), spec.getGitBranch(), roots);
                LOG.info("Space {} created from configuration with roots {}", spec.getSlug(), spec.getRoots());
            } else {
                warnOnDrift(existing.get(), spec);
            }
            LOG.info("Deploy key for space {} (add as read-only deploy key): {}", spec.getSlug(), adminService.publicKey(spec.getSlug()).trim());
        }
    }

    private void warnOnDrift(Space space, WikiProperties.SpaceSpec spec) {
        List<String> roots = rootRepository.findBySpaceWithMount(space.getId()).stream().map(r -> r.getRepoPath()).sorted().toList();
        boolean drift = !Objects.equals(space.getGitRepoUrl(), spec.getGitRepoUrl())
            || !Objects.equals(space.getGitBranch(), spec.getGitBranch())
            || !roots.equals(spec.getRoots().stream().sorted().toList());
        if (drift) {
            LOG.warn("Space {} differs from configuration (repo/branch/roots); existing space left unchanged", spec.getSlug());
        }
    }
}
