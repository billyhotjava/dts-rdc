# S1 产品身份底座 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 E3 产品集群上建立代码化的产品身份底座：`sso.dts.yuzhicloud.com`（Keycloak，realm `dts-ops` 与 `dts`），让 kubectl（apiserver OIDC）和 Rancher（迁至 `rancher.dts.yuzhicloud.com`）改用 `dts-ops` 登录，并保留可实测的应急通道。

**Architecture:** 镜像经总部 Harbor（E1 `10.20.0.50:18443`，固定名 `harbor.dts.yuzhicloud.com`）的 `mirror` 项目分发，E3 节点只经 RKE2 `registries.yaml` 拉取。平台组件按 chart-spec v1 打包为 dts-infra chart，经 `dtsctl deploy`（BOM + `DtsRelease`）在 k8s-01 上安装：`dts-gateway`（Traefik Gateway API 共享 Gateway）→ `dts-cnpg-operator` → `dts-pg`（CNPG 三实例）→ `dts-keycloak`（Keycloak Operator + 实例 + keycloak-config-cli）。节点级配置（registries、apiserver OIDC、RBAC）以脚本落在 dts-infra `deploy/e3/`，后续由 dtsctl install（T13）吸收。

**Tech Stack:** RKE2 v1.35.9、Traefik（RKE2 内置，Gateway API provider）、CloudNativePG 1.30.1、PostgreSQL 18.6、Keycloak / Keycloak Operator 26.7.5、keycloak-config-cli 6.5.1、Go 1.27.1（dtsctl）、Helm v4.2.4、crane v0.22.1、kubelogin v1.36.4、Rancher 2.15.2。

**设计依据:** [F0 模块边界与公共底座设计](../../F0-基线仓库落位与架构定案/design/2026-10-03-模块边界与公共底座设计.md)（D5、D12–D16，§5、§6 S1）；[`00`](00-dts-infra-K8s交付底座设计.md) §2、§3、§4；dts-infra `docs/chart-spec.md`、`docs/contract-spec.md`；`deploy/rancher/oidc.md`。承接 F7/T17、T19（及 T13 的 OIDC 部分、T20 的网关部分）。

```bash
#!/usr/bin/env bash
# Install the pinned E3 operator toolchain into ~/.local/dts-tools with checksum verification.
set -euo pipefail
GO_VERSION=1.27.1 KUBECTL_VERSION=v1.35.9 HELM_VERSION=v4.2.4
CRANE_VERSION=v0.22.1 KUBELOGIN_VERSION=v1.36.4
root="$HOME/.local/dts-tools"; bin="$root/bin"; cache="$root/cache"
mkdir -p "$bin" "$cache" "$HOME/.local/bin"
fetch() { [[ -s $cache/$2 ]] || curl -fsSL --retry 3 -o "$cache/$2" "$1"; }
check() { echo "$1  $cache/$2" | sha256sum -c --quiet -; }

f=go$GO_VERSION.linux-amd64.tar.gz
fetch "https://go.dev/dl/$f" "$f"
sum=$(curl -fsSL 'https://go.dev/dl/?mode=json&include=all' | python3 -c "
import json,sys; print(next(x['sha256'] for r in json.load(sys.stdin) for x in r['files'] if x['filename']=='$f'))")
check "$sum" "$f"; rm -rf "$root/go"; tar -C "$root" -xzf "$cache/$f"; ln -sf "$root/go/bin/go" "$bin/go"

fetch "https://dl.k8s.io/release/$KUBECTL_VERSION/bin/linux/amd64/kubectl" kubectl
check "$(curl -fsSL https://dl.k8s.io/release/$KUBECTL_VERSION/bin/linux/amd64/kubectl.sha256)" kubectl
install -m 755 "$cache/kubectl" "$bin/kubectl"

f=helm-$HELM_VERSION-linux-amd64.tar.gz
fetch "https://get.helm.sh/$f" "$f"
check "$(curl -fsSL https://get.helm.sh/$f.sha256sum | awk '{print $1}')" "$f"
tar -C "$cache" -xzf "$cache/$f" linux-amd64/helm; install -m 755 "$cache/linux-amd64/helm" "$bin/helm"

f=go-containerregistry_Linux_x86_64.tar.gz
fetch "https://github.com/google/go-containerregistry/releases/download/$CRANE_VERSION/$f" "crane-$f"
check "$(curl -fsSL https://github.com/google/go-containerregistry/releases/download/$CRANE_VERSION/checksums.txt | awk -v f=$f '$2==f{print $1}')" "crane-$f"
tar -C "$cache" -xzf "$cache/crane-$f" crane; install -m 755 "$cache/crane" "$bin/crane"

f=kubelogin_linux_amd64.zip
fetch "https://github.com/int128/kubelogin/releases/download/$KUBELOGIN_VERSION/$f" "$f"
check "$(curl -fsSL https://github.com/int128/kubelogin/releases/download/$KUBELOGIN_VERSION/$f.sha256 | awk '{print $1}')" "$f"
python3 -c "import zipfile; zipfile.ZipFile('$cache/$f').extract('kubelogin', '$cache')"
install -m 755 "$cache/kubelogin" "$bin/kubectl-oidc_login"

for t in go kubectl helm crane kubectl-oidc_login; do ln -sf "$bin/$t" "$HOME/.local/bin/$t"; done
go version; kubectl version --client; helm version --short; crane version; kubectl oidc-login --version
```

- [ ] **Step 1: 写 `deploy/mirror/sync.sh`**

```bash
#!/usr/bin/env bash
# Mirror upstream images into Harbor project "mirror" by digest; containerd on E3 rewrites
# <registry>/<repo> to mirror/<registry>/<repo> (see deploy/e3/registries.sh).
set -euo pipefail
harbor=${DTS_HARBOR:-10.20.0.50:18443}
export SSL_CERT_FILE=${SSL_CERT_FILE:-$HOME/.config/dts/ca-bundle.pem}

normalize() { # ref -> "registry repo tag digest"
  local ref=$1 tag="" digest="" first
  [[ $ref == *@* ]] && { digest=${ref#*@}; ref=${ref%@*}; }
  [[ ${ref##*/} == *:* ]] && { tag=${ref##*:}; ref=${ref%:*}; }
  first=${ref%%/*}
  if [[ $ref != */* || ( $first != *.* && $first != *:* && $first != localhost ) ]]; then ref=docker.io/$ref; fi
  local registry=${ref%%/*} repo=${ref#*/}
  [[ $registry == docker.io && $repo != */* ]] && repo=library/$repo
  echo "$registry $repo ${tag:--} ${digest:--}"
}

if [[ ${1:-} == --verify ]]; then
  fail=0
  while read -r line; do
    [[ -z $line || $line == \#* ]] && continue
    read -r registry repo _ digest <<<"$(normalize "$line")"
    crane digest "$harbor/mirror/$registry/$repo@$digest" >/dev/null 2>&1 || { echo "MISSING $line"; fail=1; }
  done < "${2:?lock file}"
  exit $fail
fi

list=${1:?image list}; lock=${list%.txt}.lock; : > "$lock.tmp"
while read -r line; do
  [[ -z $line || $line == \#* ]] && continue
  read -r registry repo tag digest <<<"$(normalize "$line")"
  if [[ $digest != - ]]; then
    resolved=$digest
    crane manifest "$registry/$repo@$digest" >/dev/null   # the pinned digest must exist upstream
  else
    resolved=$(crane digest "$registry/$repo:$tag")
  fi
  if [[ $tag == - ]]; then dtag=sha256-${resolved#sha256:}; else dtag=$tag; fi
  dst=$harbor/mirror/$registry/$repo:$dtag
  crane copy "$registry/$repo@$resolved" "$dst"
  echo "$registry/$repo@$resolved" >> "$lock.tmp"
done < "$list"
sort -u "$lock.tmp" > "$lock"; rm -f "$lock.tmp"
echo "locked $(wc -l < "$lock") images in $lock"
```

`charts/hack/vendor_transform.py`：

```python
#!/usr/bin/env python3
"""Convert upstream operator manifests into chart-spec v1 templates.

Namespaces become the release namespace, mapped images become values-driven image helpers,
every workload container gets the restricted securityContext, values-driven resources and a
writable /tmp, and template braces in upstream strings are escaped. Unmapped images fail.
"""
import argparse
import re
import sys

import yaml

RESTRICTED = {"allowPrivilegeEscalation": False, "readOnlyRootFilesystem": True, "runAsNonRoot": True,
              "capabilities": {"drop": ["ALL"]}, "seccompProfile": {"type": "RuntimeDefault"}}
WORKLOADS = {"Deployment", "StatefulSet", "DaemonSet"}
NS = "__DTS_NS__"


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--in", dest="src", required=True)
    p.add_argument("--out", required=True)
    p.add_argument("--helper", required=True, help="named template that renders an image reference")
    p.add_argument("--resources", required=True, help="values key holding container resources")
    p.add_argument("--image", action="append", default=[], help="UPSTREAM_REPO=VALUES_KEY")
    p.add_argument("--source-namespace", default="", help="namespace used when rendering upstream")
    a = p.parse_args()
    images = [tuple(item.split("=", 1)) for item in a.image]

    def image_token(ref):
        for i, (repo, _) in enumerate(images):
            if ref == repo or ref.startswith(repo + ":") or ref.startswith(repo + "@"):
                return "__DTS_IMAGE_%d__" % i
        return None

    def walk(node):
        if isinstance(node, dict):
            return {k: walk(v) for k, v in node.items()}
        if isinstance(node, list):
            return [walk(v) for v in node]
        if isinstance(node, str):
            if a.source_namespace and node == a.source_namespace:
                return NS
            # Upstream text must not be evaluated by Helm; restore as escaped braces after dumping.
            return node.replace("{{", "__DTS_LB__").replace("}}", "__DTS_RB__")
        return node

    text = open(a.src, encoding="utf-8").read()
    docs = []
    for doc in yaml.safe_load_all(text):
        if not doc or doc.get("kind") == "Namespace":
            continue
        doc = walk(doc)
        if doc.get("metadata", {}).get("namespace"):
            doc["metadata"]["namespace"] = NS
        for subject in doc.get("subjects") or []:
            if subject.get("kind") == "ServiceAccount" or "namespace" in subject:
                subject["namespace"] = NS
        if doc.get("kind") in WORKLOADS:
            pod = doc["spec"]["template"]["spec"]
            pod.setdefault("securityContext", {}).update(runAsNonRoot=True, seccompProfile={"type": "RuntimeDefault"})
            for c in pod.get("initContainers", []) + pod.get("containers", []):
                token = image_token(c["image"])
                if token is None:
                    sys.exit("unmapped image %s" % c["image"])
                c["image"] = token
                for env in c.get("env") or []:
                    t = image_token(str(env.get("value", "")))
                    if t:
                        env["value"] = t
                c.setdefault("securityContext", {}).update(RESTRICTED)
                c["resources"] = "__DTS_RESOURCES__"
                mounts = c.setdefault("volumeMounts", [])
                if not any(m.get("mountPath") == "/tmp" for m in mounts):
                    mounts.append({"name": "dts-tmp", "mountPath": "/tmp"})
            volumes = pod.setdefault("volumes", [])
            if not any(v.get("name") == "dts-tmp" for v in volumes):
                volumes.append({"name": "dts-tmp", "emptyDir": {}})
        docs.append(yaml.safe_dump(doc, sort_keys=False, width=1000))
    out = "---\n".join(docs).replace(NS, "{{ .Release.Namespace }}")
    out = out.replace("__DTS_LB__", '{{"{{"}}').replace("__DTS_RB__", '{{"}}"}}')
    for i, (_, key) in enumerate(images):
        out = out.replace("__DTS_IMAGE_%d__" % i,
                          '{{ include "%s" (dict "ref" .Values.%s "global" .Values.global) }}' % (a.helper, key))
    out = re.sub(r"(?m)^( *)resources: __DTS_RESOURCES__$",
                 lambda m: "%sresources: {{- toYaml .Values.%s | nindent %d }}" % (m.group(1), a.resources,
                                                                                     len(m.group(1)) + 2), out)
    with open(a.out, "w", encoding="utf-8") as handle:
        handle.write("# Generated by charts/hack/vendor_transform.py from upstream manifests; do not edit.\n")
        handle.write(out)


if __name__ == "__main__":
    main()
```

```bash
#!/usr/bin/env bash
# Master realm bootstrap: config-cli service client, break-glass admin, remove the temporary admin.
set -euo pipefail
export KUBECONFIG=/etc/rancher/rke2/rke2.yaml PATH=/var/lib/rancher/rke2/bin:$PATH
ns=dts-keycloak pod=dts-keycloak-0 out=/root/dts-identity
install -d -m 700 "$out"
rand() { python3 -c 'import secrets;print(secrets.token_urlsafe(32))'; }
kc() { kubectl -n $ns exec -i $pod -- /opt/keycloak/bin/kcadm.sh "$@" --config /tmp/kcadm.config; }
if [[ -s $out/keycloak-master-breakglass.txt ]]; then
  kc config credentials --server http://localhost:8080 --realm master --user dts-breakglass \
    --password "$(cat $out/keycloak-master-breakglass.txt)" >/dev/null
else
  user=$(kubectl -n $ns get secret dts-keycloak-initial-admin -o jsonpath='{.data.username}' | base64 -d)
  pass=$(kubectl -n $ns get secret dts-keycloak-initial-admin -o jsonpath='{.data.password}' | base64 -d)
  kc config credentials --server http://localhost:8080 --realm master --user "$user" --password "$pass" >/dev/null
  bg=$(rand); (umask 077; printf '%s\n' "$bg" > $out/keycloak-master-breakglass.txt)
  kc create users -r master -s username=dts-breakglass -s enabled=true >/dev/null
  kc set-password -r master --username dts-breakglass --new-password "$bg"
  kc add-roles -r master --uusername dts-breakglass --rolename admin
fi
if ! kubectl -n $ns get secret dts-keycloak-config-cli >/dev/null 2>&1; then
  cs=$(rand)
  kc create clients -r master -s clientId=dts-config-cli -s enabled=true -s publicClient=false \
    -s serviceAccountsEnabled=true -s standardFlowEnabled=false -s directAccessGrantsEnabled=false \
    -s secret="$cs" >/dev/null
  kc add-roles -r master --uusername service-account-dts-config-cli --rolename admin
  kubectl -n $ns create secret generic dts-keycloak-config-cli --from-literal=client-secret="$cs"
fi
# Remove the operator's temporary admin once the break-glass admin works.
kc config credentials --server http://localhost:8080 --realm master --user dts-breakglass \
  --password "$(cat $out/keycloak-master-breakglass.txt)" >/dev/null
tmp=$(kubectl -n $ns get secret dts-keycloak-initial-admin -o jsonpath='{.data.username}' | base64 -d)
id=$(kc get users -r master -q username="$tmp" -q exact=true --fields id --format csv --noquotes || true)
[[ -n $id ]] && kc delete users/$id -r master
kc get users -r master --fields username --format csv --noquotes
```

# keycloak-user.sh REALM USERNAME [GROUP] — create a user with a one-time password (UPDATE_PASSWORD).

def parse_groups(value):
